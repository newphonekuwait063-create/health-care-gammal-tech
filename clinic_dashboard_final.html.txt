  <script>
    const priceInput = document.getElementById('price');
    const monthlyLossSpan = document.getElementById('monthlyLoss');
    const annualLossSpan = document.getElementById('annualLoss');
    const missedCount = 6;

    function updateLoss() {
      const price = parseFloat(priceInput.value) || 0;
      const monthly = missedCount * price;
      const annual = monthly * 12;
      monthlyLossSpan.textContent = monthly.toLocaleString('ar-EG');
      annualLossSpan.textContent = annual.toLocaleString('ar-EG');
    }

    priceInput.addEventListener('input', updateLoss);

    function copyPost() {
      const textarea = document.getElementById('linkedinPost');
      textarea.select();
      textarea.setSelectionRange(0, 99999);
      document.execCommand('copy');
      alert('تم نسخ المنشور!');
    }
